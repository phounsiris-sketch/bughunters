# Godsmash v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rewrite badminton booking PWA from localStorage single-file to Firebase-backed multi-user app with polls, cost splitting, dinner tracking, and dashboard.

**Architecture:** Multi-file vanilla JS PWA (no build step). Firebase Auth (email link), Firestore (data), Storage (photos/QR). Files loaded via `<script>` tags. Solar icons via iconify CDN.

**Tech Stack:** Firebase 10 (CDN), Iconify (Solar icons CDN), Chart.js (dashboard charts CDN), Google Fonts (Montserrat + Noto Sans Lao)

---

## File Structure

```
pwa/
├── index.html              # App shell: HTML structure, page containers, script/css imports
├── manifest.json            # PWA manifest (update name/icons)
├── sw.js                    # Service worker (update cache list, network-first for Firestore)
├── css/
│   └── styles.css           # All styles extracted from current index.html + new pages
├── js/
│   ├── firebase-config.js   # Firebase SDK init, export db/auth/storage refs
│   ├── i18n.js              # Translation strings (en + lo) and t() helper
│   ├── auth.js              # Email link auth: login, verify, profile setup, onAuthStateChanged
│   ├── db.js                # Firestore CRUD: courts, shuttlecocks, polls, sessions, users
│   ├── storage.js           # Firebase Storage: upload QR images, dinner receipt photos
│   ├── router.js            # Page navigation, bottom tab bar, back button, auth guard
│   ├── polls.js             # Poll creation, voting UI, auto-confirm logic
│   ├── sessions.js          # Session detail, cost calculation, split display, payment marking
│   ├── dinner.js            # Dinner add/edit, receipt upload, split calculation
│   ├── dashboard.js         # Balance, spending, activity tabs with Chart.js
│   ├── settings.js          # Profile, QR management, courts/shuttlecocks/players CRUD
│   └── app.js               # Entry point: init Firebase, auth listener, load first page
└── .gitlab-ci.yml           # GitLab Pages deploy (update asset list)
```

// auth.js — Email link authentication
var currentUser = null;
var currentUserProfile = null;

var actionCodeSettings = {
  url: window.location.origin + window.location.pathname,
  handleCodeInApp: true
};

function sendLoginLink() {
  var email = document.getElementById('loginEmail').value.trim();
  if (!email) { showToast('Enter email'); return; }

  var btn = document.getElementById('sendLinkBtn');
  btn.disabled = true;

  auth.sendSignInLinkToEmail(email, actionCodeSettings)
    .then(function() {
      localStorage.setItem('emailForSignIn', email);
      document.getElementById('otpSentMsg').style.display = 'block';
      btn.disabled = false;
    })
    .catch(function(error) {
      showToast(error.message);
      btn.disabled = false;
    });
}

function handleEmailLinkSignIn() {
  if (!auth.isSignInWithEmailLink(window.location.href)) return;

  var email = localStorage.getItem('emailForSignIn');
  if (!email) {
    email = prompt('Please confirm your email:');
  }
  if (!email) return;

  auth.signInWithEmailLink(email, window.location.href)
    .then(function(result) {
      localStorage.removeItem('emailForSignIn');
      // Clean URL
      window.history.replaceState(null, '', window.location.pathname);
    })
    .catch(function(error) {
      showToast('Login failed: ' + error.message);
    });
}

function saveProfile() {
  var displayName = document.getElementById('profileName').value.trim();
  var phone = document.getElementById('profilePhone').value.trim();
  if (!displayName) { showToast('Enter name'); return; }

  db.collection('users').doc(currentUser.uid).set({
    email: currentUser.email,
    displayName: displayName,
    phone: phone || null,
    avatarUrl: null,
    createdAt: firebase.firestore.FieldValue.serverTimestamp()
  }).then(function() {
    currentUserProfile = { displayName: displayName, phone: phone, email: currentUser.email };
    showAppPage();
  }).catch(function(error) {
    showToast('Error: ' + error.message);
  });
}

function logoutUser() {
  auth.signOut();
}

function initAuth() {
  // Check for email link sign-in
  handleEmailLinkSignIn();

  // Listen for auth state changes
  auth.onAuthStateChanged(function(user) {
    if (user) {
      currentUser = user;
      // Check if profile exists
      db.collection('users').doc(user.uid).get().then(function(doc) {
        if (doc.exists) {
          currentUserProfile = doc.data();
          showAppPage();
          initApp();
        } else {
          // Show profile setup
          document.getElementById('auth-container').style.display = 'block';
          document.getElementById('app-container').style.display = 'none';
          document.getElementById('login-page').style.display = 'none';
          document.getElementById('profile-page').style.display = 'block';
        }
      });
    } else {
      currentUser = null;
      currentUserProfile = null;
      showAuthPage();
    }
  });
}

// storage.js — Firebase Storage uploads

function uploadFile(path, file, callback) {
  var ref = storage.ref().child(path);
  var task = ref.put(file);

  task.on('state_changed',
    function(snapshot) {
      var progress = Math.round((snapshot.bytesTransferred / snapshot.totalBytes) * 100);
      if (typeof callback === 'function') callback({ status: 'progress', progress: progress });
    },
    function(error) {
      showToast('Upload failed: ' + error.message);
      if (typeof callback === 'function') callback({ status: 'error', error: error });
    },
    function() {
      task.snapshot.ref.getDownloadURL().then(function(url) {
        if (typeof callback === 'function') callback({ status: 'done', url: url });
      });
    }
  );
}

function uploadQRImage(type, file, callback) {
  // type: 'courtPayer' or 'shuttlePayer'
  var path = 'qr/' + type + '_' + Date.now() + '_' + file.name;
  uploadFile(path, file, callback);
}

function uploadDinnerReceipt(sessionId, file, callback) {
  var path = 'receipts/' + sessionId + '_' + Date.now() + '_' + file.name;
  uploadFile(path, file, callback);

function toggleVote(pollId, optionIdx) {
  var pollRef = db.collection('polls').doc(pollId);

  db.runTransaction(function(transaction) {
    return transaction.get(pollRef).then(function(doc) {
      var poll = doc.data();
      if (poll.status !== 'open') return;

      var votes = poll.votes || {};
      var key = String(optionIdx);
      var optionVotes = votes[key] || [];
      var userIdx = optionVotes.indexOf(currentUser.uid);

      if (userIdx >= 0) {
        // Remove vote
        optionVotes.splice(userIdx, 1);
      } else {
        // Add vote
        optionVotes.push(currentUser.uid);
      }
      votes[key] = optionVotes;

      var updateData = { votes: votes };

      // Check if any option reached minPlayers
      if (optionVotes.length >= poll.minPlayers && userIdx < 0) {
        updateData.status = 'confirmed';
        updateData.confirmedOption = optionIdx;

        // Create session from poll
        var opt = poll.options[optionIdx];
        var sessionData = {
          pollId: pollId,
          date: opt.date,
          time: opt.time,
          courtId: opt.courtId,
          courtName: opt.courtName || opt.courtId,
          status: 'active',
          players: optionVotes,
          courtPayer: null,
          shuttlePayer: null,
          shuttlecocks: [],
          otherCosts: [],
          splits: {},
          createdAt: firebase.firestore.FieldValue.serverTimestamp()
        };

        // Create session in same transaction
        var sessionRef = db.collection('sessions').doc();
        transaction.set(sessionRef, sessionData);
        updateData.sessionId = sessionRef.id;
      }

      transaction.update(pollRef, updateData);
    });
  }).catch(function(error) {
    showToast('Vote failed: ' + error.message);
  });
}

function saveDinner(sessionId) {
  var amount = parseInt(document.getElementById('dinnerAmount').value) || 0;
  var paidBy = document.getElementById('dinnerPayerSelect').value;
  var receiptFile = document.getElementById('receiptFile').files[0];

  if (amount <= 0) { showToast(t('amount')); return; }

  var diners = [];
  document.querySelectorAll('#dinerChips .chip.active').forEach(function(chip) {
    diners.push(chip.dataset.uid);
  });
  if (diners.length === 0) { showToast(t('selectDiners')); return; }

  var splitPerPerson = Math.round(amount / diners.length);

  function saveDinnerDoc(receiptUrl) {
    dbSetDinner(sessionId, {
      totalBill: amount,
      receiptUrl: receiptUrl || null,
      diners: diners,
      splitPerPerson: splitPerPerson,
      paidBy: paidBy
    }).then(function() {
      // Update session splits to include dinner
      dbGetSession(sessionId).then(function(doc) {
        if (!doc.exists) return;
        var session = doc.data();
        var splits = session.splits || {};
        diners.forEach(function(uid) {
          if (!splits[uid]) splits[uid] = { court: 0, shuttle: 0, dinner: 0, total: 0, paid: false };
          splits[uid].dinner = splitPerPerson;
          splits[uid].total = splits[uid].court + splits[uid].shuttle + splitPerPerson;
        });
        dbUpdateSession(sessionId, { splits: splits });
      });
      closeModal();
      showToast(t('save') + '!');
    });
  }

  if (receiptFile) {
    uploadDinnerReceipt(sessionId, receiptFile, function(result) {
      if (result.status === 'done') {
        saveDinnerDoc(result.url);
      }
    });
  } else {
    saveDinnerDoc(null);
  }
}

- [ ] **Step 3: Update `sw.js`**

```javascript
var CACHE_NAME = 'godsmash-v2';
var ASSETS = [
  './',
  './index.html',
  './manifest.json',
  './css/styles.css',
  './js/firebase-config.js',
  './js/i18n.js',
  './js/router.js',
  './js/db.js',
  './js/storage.js',
  './js/auth.js',
  './js/polls.js',
  './js/sessions.js',
  './js/dinner.js',
  './js/dashboard.js',
  './js/settings.js',
  './js/app.js'
];

self.addEventListener('install', function(e) {
  e.waitUntil(
    caches.open(CACHE_NAME).then(function(cache) { return cache.addAll(ASSETS); })
  );
  self.skipWaiting();
});

self.addEventListener('activate', function(e) {
  e.waitUntil(
    caches.keys().then(function(names) {
      return Promise.all(names.filter(function(n) { return n !== CACHE_NAME; }).map(function(n) { return caches.delete(n); }));
    })
  );
  self.clients.claim();
});

self.addEventListener('fetch', function(e) {
  // Network-first for Firebase API calls
  if (e.request.url.indexOf('firestore.googleapis.com') >= 0 ||
      e.request.url.indexOf('identitytoolkit.googleapis.com') >= 0) {
    e.respondWith(fetch(e.request).catch(function() { return caches.match(e.request); }));
    return;
  }

  // Cache-first for app assets
  e.respondWith(
    caches.match(e.request).then(function(cached) {
      return cached || fetch(e.request).then(function(resp) {
        var clone = resp.clone();
        caches.open(CACHE_NAME).then(function(cache) { cache.put(e.request, clone); });
        return resp;
      });
    }).catch(function() { return caches.match('./index.html'); })
  );
});
```

## Post-Deploy: Firebase Console Setup

These steps must be done manually in Firebase Console:

1. Create project at https://console.firebase.google.com
2. Add web app, copy config to `js/firebase-config.js`
3. Authentication > Sign-in method > Enable "Email/Password" with "Email link" enabled
4. Authentication > Settings > Authorized domains > Add `phounsiri.s.gitlab.io`
5. Firestore > Create database > Start in test mode > Location: `asia-southeast1`
6. Storage > Get started > Start in test mode
7. Deploy Firestore rules from `firestore.rules`
